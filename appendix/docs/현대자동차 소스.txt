            var player;
            var playerId = idAry[i];
            player = new YT.Player(playerId, {
                width : '100%',
                height : '315',
                videoId : urlAry[i], // 페이지 전역, youtubeReady에서 설정됨

    }

    function onPlayerStateChange(event) {
        // -1:시작되지 않음, 0:종료(YT.PlayerState.ENDED), 1:재생 중(YT.PlayerState.PLAYING), 2:일시중지됨(YT.PlayerState.PAUSED), 3:버퍼링(YT.PlayerState.BUFFERING), 5:동영상 신호(YT.PlayerState.CUED)
        if(event.data == YT.PlayerState.PLAYING){
            $('#'+displayImgAry[event.target.id-1]).hide();
        }else if (event.data == YT.PlayerState.ENDED) {
            $('#mainVisual').find('.youtube-player').hide();
            //$("#storyImage").show();
            $('#'+displayImgAry[event.target.id-1]).show();
            event.target.playVideo();
        }else if(event.data == -1){
            $('#mainVisual').find('.youtube-player').hide();
            //$("#storyImage").show();
            $('#'+displayImgAry[event.target.id-1]).show();
            event.target.playVideo();
        }
    }

    function goDetail(url, selectUrlCount) {
        sessionStorage.clear();

    function getMainContents(mode) {
        mode = ComUtils.nvl(mode);
        var isFirstPage = true;
        var param = {
            langCd: GV_LANGUAGE,
            mode: mode,
            menu: menu,
            rowCnt: 12
        }
        if (mode == "MORE") { // 더보기 시
            isFirstPage = false;
            param.lastCnt = ComUtils.getValue('lastCnt'); // 최종 index
        }

        ComUtils.callAjax('/web/main/getMainContents.do', param, function (resp) {
            if (resp.result == "T") {
                if (isFirstPage) {
                    urlCount = 0;
                    urlList = [];
                }
                if (menu == 'ALL') {
                    //전체 검색일 경우 : 쿼리 조회
                    var totCnt = resp.data.resultList[0].totalCnt;
                    var html = makeBodyCardHtml(resp.data.resultList, isFirstPage, 'QR');
                    $("#mainContents").append(html);

                    ComUtils.setValue('totCnt', totCnt); // 전체 카운트 세팅

                    if (totCnt > $("div[data-id='masonryLayout'] ul li").length) {
                        $('#btnMore').show();
                    } else {
                        $('#btnMore').hide();
                    }

                    var scrollPos = Number(ComUtils.nvl(ComUtils.getCookie('goScroll'), 0));
                    ComUtils.removeCookie('goScroll');
                    if (scrollPos > 0) {
                        $(document).scrollTop(scrollPos);
                    }
                } else {
                    // 각각 메뉴의 검색일 경우 : 검색엔진

                    if (ComUtils.isEmpty(resp.data.searchList)) {
                        //검색엔진이 비어있는 경우 : 쿼리 조회된 데이터로 넣어준다.
                        var totCnt = resp.data.resultList[0].totalCnt;
                        var html = makeBodyCardHtml(resp.data.resultList, isFirstPage, 'QR');
                        $("#mainContents").append(html);
                        if (totCnt > $("div[data-id='masonryLayout'] ul li").length) {
                            $('#btnMore').show();
                        } else {
                            $('#btnMore').hide();
                        }

                        var scrollPos = Number(ComUtils.nvl(ComUtils.getCookie('goScroll'), 0));
                        ComUtils.removeCookie('goScroll');
                        if (scrollPos > 0) {
                            $(document).scrollTop(scrollPos);
                        }
                    } else {
                        //검색엔진 목록
                        var list = getSeResultList(resp.data.searchList[0].DocumentSet.Document);
                        if (!ComUtils.isEmpty(list)) {
                            totCnt = getSeResultTotalCount(resp.data.searchList);
                            var html = makeBodyCardHtml(list, isFirstPage, 'SE');
                            $("#mainContents").append(html);
                            if (totCnt > $("div[data-id='masonryLayout'] ul li").length) {
                                $('#btnMore').show();
                            } else {
                                $('#btnMore').hide();
                            }

                            var scrollPos = Number(ComUtils.nvl(ComUtils.getCookie('goScroll'), 0));
                            ComUtils.removeCookie('goScroll');
                            if (scrollPos > 0) {
                                $(document).scrollTop(scrollPos);
                            }
                        }
                    }
                }
            } else {
                alert(resp.message);
            }
        });
    }

    function getSeResultList(seResultList) {
        var resultList = [];
        if (!ComUtils.isEmpty(seResultList) && seResultList.length > 0) {
            var listCnt = seResultList.length;
            for (i = 0; i < listCnt; i++) {
                var rowFile = seResultList[i];
                var inputData = {};
                $.each(rowFile, function (key, value) {
                    inputData[ComUtils.convertCamel(key)] = value;
                });
                resultList.push(inputData);
            }
        }
        return resultList;
    }

    function makeBodyCardHtml(list, bFirstPage, mode) {
        var resultHtml = '';
        if (!ComUtils.isEmpty(list)) {
            if (bFirstPage) {
                $("#mainContents").html("");
                ComUtils.setValue('lastCnt', 0);
                list.forEach(function (item, idx) {
                    if (mode == 'SE') {
                        resultHtml += makeSeCardHtml(item.field, idx);
                    } else {
                        resultHtml += makeCardHtml(item, idx);
                    }
                    cnt = idx + 1
                });
            } else {
                list.forEach(function (item, idx) {
                    if (mode == 'SE') {
                        resultHtml += makeSeCardHtml(item.field, idx);
                    } else {
                        resultHtml += makeCardHtml(item, idx);
                    }
                    cnt = idx + 1
                })
            }
            ComUtils.setValue('lastCnt', Number(ComUtils.getValue('lastCnt')) + cnt);
        }

        return resultHtml;
    }

    function popupLayerOpen(poplinkUrl, popimgUrl, popYn, urlYn) { // popupLayerOpen(링크경로,이미지경로,1일간열지않기사용여부,url사용여부)
        var $container = $('#wrap');
        var layPopDiv = '<div class="popup-layer popup-layer-open">';

        layPopDiv += '    <div class="popup-layer__contents">';

            layPopDiv += '      </div>';
        }

        layPopDiv += '    </div>';
        layPopDiv += '    <ul class="popup-layer__buttons">';
        if (popYn == 'Y') {
            if (GV_LANGUAGE == 'KO') {
                layPopDiv += '      <li class="plb__light"><a onclick="openCheker();">1일간 열지 않기</a></li>';

        } else {
            layPopDiv += '      <li class="plb__dark"><a onclick="popupLayerClose();">close</a></li>';
        }
        layPopDiv += '    </ul>';
        layPopDiv += '  </div>';
        layPopDiv += '</div>';
        $('#wrap').prepend(layPopDiv);
        var vpageHeight = $(window).height(),
            vpopHeight = $('.popup-layer__inner').height(),
            vpopLayerTop = (vpageHeight - vpopHeight) / 2;
        if (vpopHeight < vpageHeight) {
            $('#wrap').css({'height': vpageHeight, 'overflow-y': 'hidden'});
            $('.popup-layer').css('position', 'fixed');
            $('.popup-layer__inner').css('top', vpopLayerTop);
        } else {

            $('.popup-layer').css('height', fpageHeight);
        }
    }

    function setCookie(name, value, expiredays) {
        var todayDate = new Date();
        todayDate.setDate(todayDate.getDate() + expiredays);
        document.cookie = name + "=" + escape(value) + "; path=/; expires=" + todayDate.toGMTString() + ";"
    }

    function getSmartPick(mode) {
        var param = {};
        param.langCd = GV_LANGUAGE;
        param.relatedTags = ComUtils.toJSON(ComUtils.getCookie("relatedTags"));
        param.rowCnt = 12; // 추가조회할 갯수 15
        if (mode == "MORE") { // 더보기 시
            param.lastCnt = ComUtils.getValue('lastSmartPickCnt'); // 최종 index
            param.startCount = ComUtils.getValue('lastSmartPickCnt');
        } else {
            var maxPage = Number(ComUtils.nvl(ComUtils.getCookie('maxPage'), 0));
            ComUtils.removeCookie('maxPage');
            if (maxPage > 0) {
                param.rowCnt = maxPage;
            } else {
                param.lastCnt = 0;
            }
            smartPickUrlList = new Array();
            smartPickUrlCount = 0;
            smartPickIndex = 0;
        }

        ComUtils.callAjax('/web/main/getSmartPick.do', param, function (resp) {
            if (resp.result == "T") {
                if (resp.data != null) {
                    var totSmartPickCnt = resp.data.smartPickList[0].totalCnt;
                    if (ComUtils.isEmpty(totSmartPickCnt) || totSmartPickCnt == '0') {
                        totSmartPickCnt = '0';
                    }
                    ComUtils.setValue('totSmartPickCnt', totSmartPickCnt); // 전체 카운트 세팅

                    var html = '';
                    for (var i = 0; i < resp.data.smartPickList.length; i++) {
                        var item = resp.data.smartPickList[i];
                        var pageUrl = '/' + langPath + item.pageUrl;
                        smartPickUrlList.push(pageUrl);
                        var count = smartPickUrlCount++;

                        var contentsClass = '';
                        var contentsName = '';
                        if (item.contentsType == 'VOD') {
                            contentsName = 'TV';
                        } else if (item.contentsType == 'NWS') {
                            contentsClass = 'news';
                            contentsName = '뉴스';
                        } else {
                            contentsName = '저널';
                        }

                        html += '<li class="smartPickPop__cardItem masonryLayout__item ' + contentsClass + '">';
                        html += '    <a href="' + pageUrl + '" onclick="goDetailBySmartPick(\'' + pageUrl + '\', \'' + count + '\', smartPickUrlList);event.preventDefault();" class=" smartPickPop__link masonryLayout__link">';
                        html += '        <div class="tagButtonWrap">';
                        if (item.editorPick == 'Y') {
                            html += '            <span class="tagButton tagButton--editorsPick">에디터픽</span>';
                        }
                        if (item.seriesYn == 'Y') {
                            html += '            <span class="tagButton tagButton--series">시리즈</span>';
                        }
                        html += '        </div>';
                        if ((i + 1) % 4 < 2) {
                            html += '        <div class="smartPickPop__imgWrap">';
                            html += '           <img loading="lazy" src="' + item.contentsImg + '" alt="" class="smartPickPop__img">';
                            html += '        </div>';
                        } else {
                            html += '        <div class="smartPickPop__imgWrap aspect-rectangle">';
                            html += '           <img loading="lazy" src="' + item.contentsImg + '" alt="" class="smartPickPop__img">';
                            html += '        </div>';
                        }
                        html += '        <div class="smartPickPop__textWrap">';
                        html += '            <div class="smartPickPop__title">' + item.title + '</div>';
                        html += '            <div class="smartPickPop__info">';
                        html += '                <div class="smartPickPop__info--left">';
                        html += '                    <span class="subject">' + contentsName + '</span>';
                        html += '                    <span class="date">' + item.publishDate + '</span>';
                        html += '                </div>';
                        //if(item.contentsType == 'VOD'){html += '<span class="time">'+(Math.floor((item.playTime % 3600) / 60))+':'+(item.playTime % 60)+'</span>';}
                        html += '            </div>';
                        html += '        </div>';
                        html += '    </a>';
                        html += '</li>';
                    }
                    if (mode != "MORE") { // 더보기 시
                        $('#smartPick').html('');
                    }
                    $('#smartPick').append(html);

                    smartPickIndex = smartPickIndex + resp.data.smartPickList.length;
                    ComUtils.setValue('lastSmartPickCnt', smartPickIndex);

                    if (Number(ComUtils.getValue('totSmartPickCnt')) <= Number(ComUtils.getValue('lastSmartPickCnt'))) {
                        $("#smartPickMore").hide();
                    } else {
                        $("#smartPickMore").show();
                    }
                }
            } else {
                alert(resp.message);
            }
        });
    }

    function goDetailBySmartPick(url, selectUrlCount, sessionUrlList) {
        var totUrlList = new Array();

        if (!ComUtils.isEmpty(sessionUrlList) && sessionUrlList.length > 0) {
            for (var i = 0; i < sessionUrlList.length; i++) {
                totUrlList.push(sessionUrlList[i]);

        sessionStorage.setItem("GV_STORY_FILTER_PARM", ComUtils.fromJSON(filterParam));
        sessionStorage.setItem("contentUrlList", ComUtils.fromJSON(totUrlList));
        sessionStorage.setItem("selectUrlCount", selectUrlCount);
        sessionStorage.setItem("selectTab", 'relatedTab');
        ComUtils.setCookie('contentDetailCount', '');
        ComUtils.setCookie('currentScroll', $(document).scrollTop());
        ComUtils.setCookie('maxPage', ComUtils.getValue('lastCnt'));
        ComUtils.setCookie('webReferrer', document.location.href);
        ComUtils.redirect(url);
    }

                    alert("Enter your email address.");
                    $("input[id='email']").focus();
                    return false;

    function getFooterGroupList(){
        var param = {};
        param.langCd = ComUtils.getLang();

        ComUtils.callAjax('/web/gnb/getFooterGroupList.do', param, function(resp){
            if(resp.result == "T") {
                var groupList = resp.data;
                var html = '';
                if(!ComUtils.isEmpty(groupList)){
                    html += '<ul class="hmg-site__list">';
                    for(var i=0; i<groupList.length; i++){
                        var data = groupList[i];
                        //html += '<li><a href="'+data.groupLink+'" target="_blank" rel="noopener noreferrer" title="'+data.groupNm+'">'+data.groupNm+'</a></li>';
                        html += '<li><a href="'+data.groupLink+'" target="_blank" title="'+GV_LINK_BLANK_TITLE+'" rel="noopener noreferrer">'+data.groupNm+'</a></li>';
                    }

                    html += '</ul>';

                    $("#footerGroupList").append(html);
                }
            } else {
                alert(resp.message);
            }
        });
    }
